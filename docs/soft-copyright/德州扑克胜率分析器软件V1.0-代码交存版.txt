22     function getParallelPlan(totalSimulations, numOpponents) {
23         const cores = Math.max(1, Number(navigator.hardwareConcurrency) || 4);
24         const maxWorkers = Math.max(1, Math.min(8, cores - 1 || 1));
25 
26         let preferredWorkers = 2;
27         if (numOpponents >= 7) preferredWorkers = 6;
28         else if (numOpponents >= 5) preferredWorkers = 5;
29         else if (numOpponents >= 3) preferredWorkers = 4;
30         else preferredWorkers = 3;
31 
32         const minSimsPerWorker = numOpponents >= 6 ? 2200 : 3000;
33         const capByWorkload = Math.max(1, Math.floor(totalSimulations / minSimsPerWorker));
34         const workerCount = Math.max(1, Math.min(preferredWorkers, maxWorkers, capByWorkload));
35 
36         const chunks = [];
37         let remaining = totalSimulations;
38         for (let i = 0; i < workerCount; i++) {
39             const chunk = Math.ceil(remaining / (workerCount - i));
40             chunks.push(chunk);
41             remaining -= chunk;
42         }
43

24 function extractResultText(response) {

30         if (choice.message && typeof choice.message.content === 'string') {

34 
35     // Workers AI native format
36     if (typeof response.response === 'string') return response.response.trim();
37     if (typeof response.result === 'string')   return response.result.trim();

39     // Legacy Image-to-Text
40     if (typeof response.description === 'string') return response.description.trim();

43     if (typeof response === 'string') return response.trim();

49  * Extract only the final answer line from CoT output.
50  * Models may produce reasoning text before the actual card codes.

01  * We look for a line that matches the expected card code pattern.
02  */
03 function extractCardCodesFromCoT(rawText) {
04     const text = String(rawText || '').trim();
05     if (!text) return text;
06 
07     // If the entire text is short and looks like card codes, return as-is
08     const cardPattern = /^(?:None|(?:[2-9TJQKA][shdc](?:\s*,\s*)?)+)$/i;
09     if (cardPattern.test(text)) return text;
10 
11     // Look for the last line that contains card codes
12     const lines = text.split('\n').map(l => l.trim()).filter(Boolean);
13     for (let i = lines.length - 1; i >= 0; i--) {
14         const line = lines[i];
15         // Match lines like "Ah, Kd, 9c" or "Ah,Kd,9c"
16         if (/^[2-9TJQKA][shdc](\s*,\s*[2-9TJQKA][shdc])*$/i.test(line)) {
17             return line;
18         }
19         // Match lines starting with "Answer:" or "Result:" or "Output:" prefix
20         const prefixMatch = line.match(/^(?:answer|result|output|final|cards?)\s*[:：]\s*(.+)$/i);
21         if (prefixMatch) {
22             return prefixMatch[1].trim();
23         }
24     }
25 
26     // Fallback: return original text — the frontend parser will handle it
27     return text;
28 }
